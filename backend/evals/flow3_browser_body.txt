function gate(scene){const check=netCheck(scene),poly=polyhedronOf(scene);let edges={};if(poly){for(const f of poly.faces)for(let i=0;i<f.vertices.length;i++){const key=[f.vertices[i],f.vertices[(i+1)%f.vertices.length]].sort((a,b)=>a-b).join('-');edges[key]=(edges[key]||0)+1}}const manifold=!poly||Object.values(edges).every(n=>n===2);return {check,faces:poly?.faces.length,vertices:poly?.positions.length,edges:Object.keys(edges).length,manifold,valid:(!check||check.closed)&&manifold};}
